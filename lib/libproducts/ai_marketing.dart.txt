import 'package:flutter/material.dart';

class AIMarketingScreen extends StatelessWidget {
  final Map<String, dynamic> product;
  AIMarketingScreen({required this.product});

  void generateAIAds(BuildContext context) {
    // Placeholder for AI image/video marketing generation
    ScaffoldMessenger.of(context)
        .showSnackBar(SnackBar(content: Text("AI Marketing assets generated!")));
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: Text("AI Marketing for ${product['name']}")),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          children: [
            product['image'] != ""
                ? Image.asset(product['image'], height: 200)
                : Container(height: 200, color: Colors.grey),
            SizedBox(height: 20),
            Text("Category: ${product['category']}"),
            Text("Price: ₹${product['price']}"),
            Text("Description: ${product['description']}"),
            SizedBox(height: 20),
            ElevatedButton(
                onPressed: () => generateAIAds(context),
                child: Text("Generate AI Marketing Assets"))
          ],
        ),
      ),
    );
  }
}
